using System;
using System.IO;
using System.Reflection;
using System.Runtime.InteropServices;

namespace ConsoleApplication4
{
	/// <summary>
	/// Summary description for Class1.
	/// </summary>
	class Class1
	{
		/// <summary>
		/// The main entry point for the application.
		/// </summary>
		[STAThread]
		static void Main(string[] args)
		{
			Type fIT = Type.GetTypeFromProgID("IMAPI2FS.MsftFileSystemImage");
			object oFI = Activator.CreateInstance(fIT);
			fIT.InvokeMember("ChooseImageDefaultsForMediaType", BindingFlags.InvokeMethod, null, oFI, new object[] { 3 });
			object rootfldr = fIT.InvokeMember("Root", BindingFlags.GetProperty, null, oFI, null);
			object[] addTree = new object[] {@"C:\SRC", false};
			rootfldr.GetType().InvokeMember("AddTree",BindingFlags.InvokeMethod,null,rootfldr,addTree);
			object resultImage = fIT.InvokeMember("CreateResultImage",BindingFlags.InvokeMethod,null,oFI,null);
			UCOMIStream comStream = (UCOMIStream) resultImage.GetType().InvokeMember("ImageStream",BindingFlags.GetProperty,null,resultImage,null);
			FileStream fs = new FileStream(@"C:\lol.iso",FileMode.Create,FileAccess.Write);
			byte[] buffer = new byte[4096];
			int bytesRead = 0;
			IntPtr pcbRead = Marshal.AllocCoTaskMem(Marshal.SizeOf(typeof(int)));
			try
			{
				while (true)
				{
					comStream.Read(buffer,buffer.Length,pcbRead);
					bytesRead = Marshal.ReadInt32(pcbRead);
					if (bytesRead == 0)
					{
						break;
					}
					fs.Write(buffer,0,bytesRead);
				}
			}
			finally
			{
				Marshal.FreeCoTaskMem(pcbRead);
				fs.Flush();
				fs.Close();
			}
		}
	}
}
